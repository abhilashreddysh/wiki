```yaml
core_services:
- name: DNS / DHCP
    type: network services
    restore_priority: critical
    hosted_on: gateway / virtual node
    dependency: network-online.target

- name: VPN Mesh
    type: encrypted overlay
    restore_priority: high
    hosted_on: gateway / virtual node
    dependency: network-online.target

- name: File Sharing
    type: storage service
    restore_priority: high
    hosted_on: workstation
    dependency: DNS / DHCP
```
